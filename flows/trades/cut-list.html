/* Cut List. * * First-fit-decreasing bin packing: sort the pieces longest first, and put each * one in the first bar it still fits. It is not guaranteed optimal — nothing * fast is — but it is within a bar or two of optimal on real cutting lists, and * it produces a layout a person can follow down the bench in order. * * The blade width is not a detail. Ten cuts with a 3 mm blade is 30 mm gone, * which is exactly the amount that makes the last piece not fit. */ Flow.ready(function () { var UNITS = { mm: { step: 1, dp: 0, kerf: 3 }, m: { step: 0.001, dp: 3, kerf: 0.003 }, in: { step: 0.0625, dp: 3, kerf: 0.125 } }; var s = Flow.state('cuts', { unit: 'mm', stock: '2400', kerf: '3', pieces: [ { label: 'Shelf', length: '800', qty: '4' }, { label: 'Upright', length: '1800', qty: '2' } ] }); var el = { stock: Flow.$('#stock'), kerf: Flow.$('#kerf'), unit: Flow.$('#unit'), pieces: Flow.$('#pieces'), bars: Flow.$('#bars'), answer: Flow.$('#answer'), note: Flow.$('#note'), block: Flow.$('#plan-block') }; var num = function (v) { var n = parseFloat(v); return isFinite(n) && n > 0 ? n : 0; }; var fmt = function (n) { return Flow.num(n, UNITS[s.unit].dp).replace(/\.?0+$/, function (m) { return s.unit === 'mm' ? '' : m; }); }; var label = function (n) { return fmt(n) + ' ' + s.unit; }; function pack() { var stock = num(s.stock); var kerf = parseFloat(s.kerf) || 0; var wanted = []; s.pieces.forEach(function (p) { var len = num(p.length); var qty = Math.max(0, Math.round(num(p.qty))); for (var i = 0; i < qty && i < 400; i++) wanted.push({ len: len, label: p.label }); }); var tooLong = wanted.filter(function (p) { return p.len > stock; }); var fits = wanted.filter(function (p) { return p.len <= stock && p.len > 0; }); fits.sort(function (a, b) { return b.len - a.len; }); var bars = []; fits.forEach(function (piece) { // A cut is only charged for the blade if something has already been cut // off this bar, so the first piece on a fresh length is kerf-free. var home = bars.find(function (bar) { return bar.left >= piece.len + (bar.pieces.length ? kerf : 0); }); if (!home) { home = { pieces: [], left: stock }; bars.push(home); } home.left -= piece.len + (home.pieces.length ? kerf : 0); home.pieces.push(piece); }); var used = fits.reduce(function (a, p) { return a + p.len; }, 0); return { stock: stock, bars: bars, tooLong: tooLong, used: used, bought: bars.length * stock, waste: bars.length ? (1 - used / (bars.length * stock)) * 100 : 0, longestOff: bars.reduce(function (a, b) { return Math.max(a, b.left); }, 0) }; } function renderPieces() { el.pieces.innerHTML = ''; s.pieces.forEach(function (p, i) { var row = Flow.h('div', { class: 'cl-row' }); row.appendChild(Flow.h('input', { class: 'flow-input', value: p.label, placeholder: 'Piece ' + (i + 1), 'aria-label': 'What for', oninput: function (e) { p.label = e.target.value; save(); update(); } })); row.appendChild(Flow.h('input', { class: 'flow-input len', type: 'text', inputmode: 'decimal', value: p.length, placeholder: '0', 'aria-label': 'Length', oninput: function (e) { p.length = e.target.value; save(); update(); } })); row.appendChild(Flow.h('input', { class: 'flow-input qty', type: 'text', inputmode: 'numeric', value: p.qty, placeholder: '1', 'aria-label': 'How many', oninput: function (e) { p.qty = e.target.value; save(); update(); } })); row.appendChild(Flow.h('button', { class: 'flow-x', type: 'button', 'aria-label': 'Remove', html: Flow.icons.x, onclick: function () { s.pieces.splice(i, 1); s.save(); renderPieces(); update(); } })); el.pieces.appendChild(row); }); } function update() { var r = pack(); el.block.classList.toggle('flow-hidden', !r.bars.length && !r.tooLong.length); el.answer.innerHTML = ''; var tile = Flow.h('div', { class: 'flow-tile' }); tile.appendChild(Flow.h('div', { class: 'flow-tile-label', text: 'Lengths to buy' })); tile.appendChild(Flow.h('div', { class: 'flow-tile-value', text: String(r.bars.length) })); tile.appendChild(Flow.h('div', { class: 'flow-tile-note', text: r.bars.length ? 'at ' + label(r.stock) + ' each · ' + Flow.num(r.waste, 1) + '% offcut · longest offcut ' + label(r.longestOff) : 'Add the pieces you need.' })); el.answer.appendChild(tile); el.bars.innerHTML = ''; r.bars.forEach(function (bar, i) { var wrap = Flow.h('div', { class: 'cl-bar' }); var top = Flow.h('div', { class: 'cl-bar-top' }); top.appendChild(Flow.h('b', { text: 'Length ' + (i + 1) })); top.appendChild(Flow.h('span', { text: bar.left > 0.0001 ? label(bar.left) + ' left' : 'nothing left' })); wrap.appendChild(top); var stick = Flow.h('div', { class: 'cl-stick' }); bar.pieces.forEach(function (p) { stick.appendChild(Flow.h('div', { class: 'cl-piece', style: { flex: p.len + ' 1 0' }, title: p.label, text: fmt(p.len) })); }); if (bar.left > 0.0001) { stick.appendChild(Flow.h('div', { class: 'cl-off', style: { flex: bar.left + ' 1 0' }, text: bar.left / r.stock > 0.12 ? fmt(bar.left) : '' })); } wrap.appendChild(stick); el.bars.appendChild(wrap); }); el.note.textContent = r.tooLong.length ? r.tooLong.length + ' piece' + (r.tooLong.length === 1 ? ' is' : 's are') + ' longer than the stock — they cannot be cut from it.' : (r.bars.length ? 'Cut each length in the order shown, longest first. The striped part is offcut.' : ''); el.note.style.color = r.tooLong.length ? 'var(--danger)' : ''; } function applyUnit() { el.unit.value = s.unit; Flow.$('#stock-unit').textContent = s.unit; Flow.$('#kerf-unit').textContent = s.unit; } var timer; function save() { clearTimeout(timer); timer = setTimeout(function () { s.save(); }, 300); } el.stock.value = s.stock; el.kerf.value = s.kerf; applyUnit(); el.stock.addEventListener('input', function (e) { s.stock = e.target.value; save(); update(); }); el.kerf.addEventListener('input', function (e) { s.kerf = e.target.value; save(); update(); }); el.unit.addEventListener('change', function (e) { // Units change the labels, not the numbers: retyping a whole cut list // because you picked the wrong unit first is the sort of thing that makes // people close a tool and reach for paper. s.unit = e.target.value; s.kerf = String(UNITS[s.unit].kerf); el.kerf.value = s.kerf; s.save(); applyUnit(); update(); }); Flow.$('#add').addEventListener('click', function () { s.pieces.push({ label: '', length: '', qty: '1' }); s.save(); renderPieces(); update(); var inputs = Flow.$$('.cl-row .flow-input'); if (inputs.length) inputs[inputs.length - 3].focus(); }); renderPieces(); update(); });
