/* Idea Board. * * The only thing that matters here is the gap between having a thought and it * being safe: one box, always at the top, always focusable, saving as you type * so a phone dying mid-sentence costs nothing. Tags are read out of the text * rather than entered separately, because a second field is a second decision * and the thought does not survive it. */ Flow.ready(function () { var s = Flow.state('ideas', { items: [], filter: null }); var draft = Flow.$('#draft'); var keep = Flow.$('#keep'); var listEl = Flow.$('#list'); var filtersEl = Flow.$('#filters'); var tagsIn = function (text) { var found = (text.match(/#[\w-]{1,24}/g) || []).map(function (t) { return t.slice(1).toLowerCase(); }); return found.filter(function (t, i) { return found.indexOf(t) === i; }); }; function when(iso) { var mins = Math.round((Date.now() - new Date(iso).getTime()) / 60000); if (mins < 1) return 'just now'; if (mins < 60) return mins + ' min ago'; var hrs = Math.round(mins / 60); if (hrs < 24) return hrs + (hrs === 1 ? ' hour ago' : ' hours ago'); var days = Math.round(hrs / 24); if (days < 7) return days + (days === 1 ? ' day ago' : ' days ago'); return new Date(iso).toLocaleDateString(undefined, { day: 'numeric', month: 'short' }); } function grow() { draft.style.height = 'auto'; draft.style.height = Math.min(draft.scrollHeight, 260) + 'px'; keep.disabled = !draft.value.trim(); } function add() { var text = draft.value.trim(); if (!text) return; s.items.unshift({ id: String(Date.now()) + Math.random().toString(36).slice(2, 6), text: text, tags: tagsIn(text), created: new Date().toISOString(), pinned: false }); s.save(); Flow.store.del('draft'); draft.value = ''; grow(); render(); Flow.haptic(); } function render() { var all = s.items.slice().sort(function (a, b) { if (!!a.pinned !== !!b.pinned) return a.pinned ? -1 : 1; return b.created.localeCompare(a.created); }); var shown = s.filter ? all.filter(function (i) { return i.tags.indexOf(s.filter) > -1; }) : all; renderFilters(all); listEl.innerHTML = ''; if (!shown.length) { listEl.appendChild(Flow.h('div', { class: 'flow-empty' }, Flow.h('strong', { text: s.filter ? 'Nothing tagged #' + s.filter : 'Nothing here yet' }), s.filter ? 'Try another tag.' : 'The box above is the whole app. Type and forget.')); return; } shown.forEach(function (item) { var card = Flow.h('div', { class: 'ib-item' + (item.pinned ? ' is-pinned' : '') }); card.appendChild(Flow.h('p', { text: item.text })); var foot = Flow.h('div', { class: 'ib-foot' }); foot.appendChild(Flow.h('span', { class: 'ib-when', text: when(item.created) })); item.tags.forEach(function (t) { foot.appendChild(Flow.h('button', { class: 'ib-tag', type: 'button', text: '#' + t, onclick: function () { s.filter = s.filter === t ? null : t; s.save(); render(); } })); }); foot.appendChild(Flow.h('button', { class: 'flow-x', type: 'button', 'aria-label': 'More', html: Flow.icons.more, onclick: function () { menu(item); } })); card.appendChild(foot); if (item.pinned) card.appendChild(Flow.h('span', { class: 'ib-pin', html: Flow.icons.check, style: { color: 'var(--a2)' } })); listEl.appendChild(card); }); } function renderFilters(all) { var counts = {}; all.forEach(function (i) { i.tags.forEach(function (t) { counts[t] = (counts[t] || 0) + 1; }); }); var tags = Object.keys(counts).sort(function (a, b) { return counts[b] - counts[a] || a.localeCompare(b); }); filtersEl.classList.toggle('flow-hidden', tags.length < 2); filtersEl.innerHTML = ''; if (tags.length < 2) return; filtersEl.appendChild(Flow.h('button', { class: 'flow-chip', type: 'button', 'aria-pressed': s.filter ? 'false' : 'true', text: 'Everything', onclick: function () { s.filter = null; s.save(); render(); } })); tags.slice(0, 14).forEach(function (t) { filtersEl.appendChild(Flow.h('button', { class: 'flow-chip', type: 'button', 'aria-pressed': s.filter === t ? 'true' : 'false', text: '#' + t + ' ' + counts[t], onclick: function () { s.filter = s.filter === t ? null : t; s.save(); render(); } })); }); } function menu(item) { var handle = Flow.sheet({ title: item.pinned ? 'Pinned idea' : 'Idea', lede: item.text.length > 90 ? item.text.slice(0, 90) + '…' : item.text, actions: [ { label: item.pinned ? 'Unpin it' : 'Pin to the top', kind: 'quiet', onClick: function () { item.pinned = !item.pinned; s.save(); render(); } }, { label: 'Copy the text', kind: 'quiet', onClick: function () { if (navigator.clipboard) navigator.clipboard.writeText(item.text).then(function () { Flow.toast('Copied'); }); } }, { label: 'Put it back in the box', kind: 'quiet', onClick: function () { draft.value = item.text; grow(); draft.focus(); s.items = s.items.filter(function (i) { return i.id !== item.id; }); s.save(); render(); } }, { label: 'Delete', kind: 'danger', onClick: function () { s.items = s.items.filter(function (i) { return i.id !== item.id; }); s.save(); render(); Flow.toast('Gone'); } }, { label: 'Cancel', kind: 'ghost' } ] }); return handle; } // An unsent draft survives the app closing.
