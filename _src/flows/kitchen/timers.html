/* Kitchen Timers. * * Several at once, because a meal is several things at once. Each timer stores * the wall-clock moment it finishes rather than a countdown, so closing the * app, locking the phone or reloading the page all leave the rice exactly as * far along as it really is. */ Flow.ready(function () { var PRESETS = [ ['Soft egg', 6 * 60 + 30], ['Hard egg', 9 * 60], ['Pasta', 10 * 60], ['Rice', 12 * 60], ['Tea', 3 * 60], ['Steak, rest', 5 * 60], ['Toast', 3 * 60], ['Oven check', 15 * 60] ]; var s = Flow.state('timers', { list: [] }); var listEl = Flow.$('#list'); var alarming = {}; var uid = function () { return String(Date.now()) + Math.random().toString(36).slice(2, 5); }; function leftOn(t) { if (t.endsAt) return Math.max(0, (t.endsAt - Date.now()) / 1000); return t.paused !== null && t.paused !== undefined ? t.paused : t.total; } var isRunning = function (t) { return !!t.endsAt && t.endsAt > Date.now(); }; var isDone = function (t) { return !!t.endsAt && t.endsAt <= Date.now(); }; var PLAY = ''; var PAUSE = ''; var STOP = ''; function render() { listEl.innerHTML = ''; if (!s.list.length) { listEl.appendChild(Flow.h('div', { class: 'flow-empty' }, Flow.h('strong', { text: 'No timers running' }), 'Add one for each pan. They count down together.')); return; } s.list.forEach(function (t) { var done = isDone(t); var running = isRunning(t); var left = leftOn(t); var card = Flow.h('div', { class: 'kt-card' + (done ? ' is-done' : running ? ' is-running' : ''), dataset: { id: t.id }, style: { '--done': (t.total > 0 ? (1 - left / t.total) * 100 : 0) + '%' } }); card.appendChild(Flow.h('button', { class: 'kt-play', type: 'button', 'aria-label': done ? 'Stop' : running ? 'Pause' : 'Start', html: done ? STOP : running ? PAUSE : PLAY, onclick: function () { toggle(t); } })); var body = Flow.h('div', { class: 'kt-body' }); body.appendChild(Flow.h('div', { class: 'kt-name', text: t.name })); body.appendChild(Flow.h('div', { class: 'kt-left', text: done ? 'Time’s up' : Flow.dur(left) })); card.appendChild(body); card.appendChild(Flow.h('button', { class: 'flow-x', type: 'button', 'aria-label': 'Remove', html: Flow.icons.x, onclick: function () { remove(t); } })); if (done && !alarming[t.id]) startAlarm(t); listEl.appendChild(card); }); } // Repainting only the numbers keeps the tick cheap and, more usefully, keeps // a card from being rebuilt out from under a finger mid-tap. function paint() { var restructure = false; s.list.forEach(function (t) { var card = listEl.querySelector('[data-id="' + t.id + '"]'); if (!card) { restructure = true; return; } var done = isDone(t); var left = leftOn(t); card.style.setProperty('--done', (t.total > 0 ? (1 - left / t.total) * 100 : 0) + '%'); var label = card.querySelector('.kt-left'); var next = done ? 'Time’s up' : Flow.dur(left); if (label.textContent !== next) label.textContent = next; // A timer crossing zero changes the card's buttons, not just its number. if (card.className !== 'kt-card' + (done ? ' is-done' : isRunning(t) ? ' is-running' : '')) restructure = true; if (done && !alarming[t.id]) startAlarm(t); }); if (restructure) render(); } function startAlarm(t) { alarming[t.id] = setInterval(function () { Flow.beep({ count: 2, freq: 1046, gap: 0.2, length: 0.16, volume: 0.28 }); Flow.haptic(120); }, 2000); Flow.beep({ count: 3, freq: 1046, gap: 0.2, length: 0.16, volume: 0.3 }); Flow.haptic(180); Flow.keepAwake(false); } function stopAlarm(id) { if (alarming[id]) { clearInterval(alarming[id]); delete alarming[id]; } } function toggle(t) { if (isDone(t)) { stopAlarm(t.id); t.endsAt = null; t.paused = t.total; } else if (isRunning(t)) { t.paused = leftOn(t); t.endsAt = null; } else { Flow.audio(); t.endsAt = Date.now() + leftOn(t) * 1000; t.paused = null; } s.save(); syncWake(); render(); Flow.haptic(); } function remove(t) { stopAlarm(t.id); s.list = s.list.filter(function (x) { return x.id !== t.id; }); s.save(); syncWake(); render(); } function syncWake() { Flow.keepAwake(s.list.some(isRunning)); } function addSheet() { var mins = 10; var secs = 0; var name = ''; var body = Flow.h('div'); var presets = Flow.h('div', { class: 'kt-presets' }); PRESETS.forEach(function (p) { presets.appendChild(Flow.h('button', { class: 'kt-preset', type: 'button', onclick: function () { handle.close(); create(p[0], p[1]); } }, Flow.h('b', { text: p[0], style: { fontWeight: '550' } }), Flow.h('span', { text: Flow.dur(p[1]) }))); }); body.appendChild(presets); body.appendChild(Flow.h('span', { class: 'flow-label', text: 'Or set your own' })); var nameInput = Flow.h('input', { class: 'flow-input', placeholder: 'What is it for', 'aria-label': 'Name', oninput: function (e) { name = e.target.value; } }); body.appendChild(nameInput); var dials = Flow.h('div', { class: 'kt-dials' }); function dial(label, get, set, step, max) { var out = Flow.h('output', { text: String(get()) }); var wrap = Flow.h('div'); wrap.appendChild(Flow.h('span', { class: 'flow-label', text: label, style: { marginTop: '10px' } })); wrap.appendChild(Flow.h('div', { class: 'flow-stepper' }, Flow.h('button', { type: 'button', text: '−', 'aria-label': 'Less', onclick: function () { set(Math.max(0, get() - step)); out.textContent = get(); Flow.haptic(); } }), out, Flow.h('button', { type: 'button', text: '+', 'aria-label': 'More', onclick: function () { set(Math.min(max, get() + step)); out.textContent = get(); Flow.haptic(); } }) )); dials.appendChild(wrap); } dial('Minutes', function () { return mins; }, function (v) { mins = v; }, 1, 180); dial('Seconds', function () { return secs; }, function (v) { secs = v; }, 15, 45); body.appendChild(dials); var handle = Flow.sheet({ title: 'Add a timer', lede: 'Pick something common, or dial in your own.', body: body, actions: [ { label: 'Start it', onClick: function () { create(name, mins * 60 + secs); } }, { label: 'Cancel', kind: 'ghost' } ] }); } function create(name, seconds) { if (!(seconds > 0)) return Flow.toast('That timer has no time in it'); Flow.audio(); s.list.push({ id: uid(), name: (name || '').trim() || Flow.dur(seconds) + ' timer', total: seconds, endsAt: Date.now() + seconds * 1000, paused: null }); s.save(); syncWake(); render(); Flow.haptic(30); } Flow.$('#add').addEventListener('click', addSheet); setInterval(paint, 250); document.addEventListener('visibilitychange', function () { if (document.visibilityState === 'visible') paint(); }); syncWake(); render(); });
